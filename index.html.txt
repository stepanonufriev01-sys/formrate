    <script>
        let currentLang = 'ru';
        let ratingsGivenCount = 0;
        const targetRatings = 3;

        const translations = {
            ru: {
                height: "Рост", weight: "Вес", exp: "Стаж",
                questTitle: "Чтобы опубликовать форму, оцени участников:",
                ratePrompt: "Оцени форму (1-10):",
                sendRateBtn: "Отправить оценку и далее →",
                myProfileTitle: "Твоя Анкета Формы",
                expLabel: "Стаж тренировок",
                photoLabel: "Фото формы (в полный рост)",
                publishBtn: "Опубликовать мою анкету ⚡",
                lockText: "Оцените еще участников, чтобы разблокировать загрузку своей формы!",
                proDesc: "Получи максимум от сервиса и выведи свои результаты на новый уровень."
            },
            en: {
                height: "Height", weight: "Weight", exp: "Exp",
                questTitle: "Rate members to publish your form:",
                ratePrompt: "Rate form (1-10):",
                sendRateBtn: "Submit rating & next →",
                myProfileTitle: "Your Form Profile",
                expLabel: "Training Experience",
                photoLabel: "Full body photo",
                publishBtn: "Publish My Form ⚡",
                lockText: "Rate more members to unlock your profile upload!",
                proDesc: "Get maximum features and take your physique to the next level."
            }
        };

        function switchLang() {
            currentLang = currentLang === 'ru' ? 'en' : 'ru';
            document.getElementById('langBtn').innerText = currentLang.toUpperCase();
            
            document.querySelectorAll('[data-i18n]').forEach(el => {
                const key = el.getAttribute('data-i18n');
                if (translations[currentLang][key]) {
                    el.innerText = translations[currentLang][key];
                }
            });
        }

        function submitRating() {
            ratingsGivenCount++;
            updateQuestProgress();
            
            // Заглушка переключения на следующую карточку
            document.getElementById('rateHeight').innerText = (175 + Math.floor(Math.random() * 10)) + ' см';
            document.getElementById('rateWeight').innerText = (75 + Math.floor(Math.random() * 15)) + ' кг';
            document.getElementById('cardBadge').innerText = `Атлет #${ratingsGivenCount + 1}`;
        }

        function updateQuestProgress() {
            const left = targetRatings - ratingsGivenCount;
            document.getElementById('questCounter').innerText = `${ratingsGivenCount} / ${targetRatings}`;
            document.getElementById('questBar').style.width = `${(ratingsGivenCount / targetRatings) * 100}%`;
            document.getElementById('leftCount').innerText = left > 0 ? left : 0;

            if (ratingsGivenCount >= targetRatings) {
                // Разблокируем форму
                const uploadSec = document.getElementById('uploadSection');
                uploadSec.classList.remove('opacity-50', 'pointer-events-none');
                document.getElementById('lockOverlay').style.display = 'none';
                document.getElementById('questBox').classList.add('border-green-500');
                document.getElementById('questTitle').innerText = currentLang === 'ru' ? '✅ Цель выполнена! Загружайте свою форму:' : '✅ Goal met! Upload your form now:';
            }
        }

        async function handleFormSubmit(event) {
            event.preventDefault();
            const height = document.getElementById('myHeight').value;
            const weight = document.getElementById('myWeight').value;
            const exp = document.getElementById('myExp').value;
            const photoFile = document.getElementById('myPhoto').files[0];

            if (!photoFile) return alert('Загрузите фото!');

            try {
                // 1. Загрузка фото в Supabase Storage
                const fileName = `${Date.now()}_${photoFile.name}`;
                const { data: uploadData, error: uploadError } = await supabaseClient.storage
                    .from('form-photos')
                    .upload(fileName, photoFile);

                if (uploadError) throw uploadError;

                // Получаем публичную ссылку на фото
                const { data: publicUrlData } = supabaseClient.storage
                    .from('form-photos')
                    .getPublicUrl(fileName);

                const photoUrl = publicUrlData.publicUrl;

                // 2. Сохранение данных профиля в базу данных Supabase
                const { error: dbError } = await supabaseClient
                    .from('profiles')
                    .insert([{ height, weight, experience: exp, photo_url: photoUrl, ratings_given: ratingsGivenCount }]);

                if (dbError) throw dbError;

                alert(currentLang === 'ru' ? '⚡ Анкета успешно опубликована в базе FormRate!' : '⚡ Profile successfully published!');
                location.reload();

            } catch (err) {
                console.error(err);
                alert('Ошибка при сохранении: ' + err.message);
            }
        }

        function openProModal() { document.getElementById('proModal').classList.remove('hidden'); }
        function closeProModal() { document.getElementById('proModal').classList.add('hidden'); }
    </script>
